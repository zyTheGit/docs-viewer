## 路由

### 路由前缀

```typescript
@Controller('users')  // 路由前缀: /users
export class UsersController {
  @Get()              // GET /users
  findAll() {}

  @Get(':id')         // GET /users/:id
  findOne(@Param('id') id: string) {}
}

@Controller('admin/users')  // 路由前缀: /admin/users
export class AdminUsersController {}
```

## 请求方法

```typescript
@Controller('users')
export class UsersController {
  @Get()           // GET /users
  findAll() {}

  @Post()          // POST /users
  create() {}

  @Put(':id')      // PUT /users/:id
  update() {}

  @Patch(':id')    // PATCH /users/:id
  partialUpdate() {}

  @Delete(':id')   // DELETE /users/:id
  remove() {}

  @Options()       // OPTIONS /users
  options() {}

  @Head()          // HEAD /users
  head() {}

  @All()           // 所有 HTTP 方法
  all() {}
}
```

## 参数装饰器

### @Body - 请求体

```typescript
@Post()
create(@Body() createUserDto: CreateUserDto) {
  return this.usersService.create(createUserDto);
}

@Post()
create(@Body('name') name: string, @Body('email') email: string) {
  return { name, email };
}
```

### @Param - 路由参数

```typescript
@Get(':id')
findOne(@Param('id') id: string) {
  return this.usersService.findOne(id);
}

@Get(':id/posts/:postId')
findPost(@Param('id') id: string, @Param('postId') postId: string) {
  return { userId: id, postId };
}

@Get(':id')
findOne(@Param() params: { id: string }) {
  return params.id;
}
```

### @Query - 查询参数

```typescript
@Get()
findAll(
  @Query('page') page: number = 1,
  @Query('limit') limit: number = 10,
) {
  return this.usersService.findAll({ page, limit });
}

@Get()
findAll(@Query() query: any) {
  return query;
}
```

### @Headers - 请求头

```typescript
@Get()
findAll(@Headers('authorization') token: string) {
  return { token };
}

@Get()
findAll(@Headers() headers: Record<string, string>) {
  return headers;
}
```

### @Ip - 客户端 IP

```typescript
@Get()
findAll(@Ip() ip: string) {
  return { ip };
}
```

### @Session - Session

```typescript
@Get()
findAll(@Session() session: Record<string, any>) {
  return session;
}
```

### @Req / @Res - 原始请求/响应对象

```typescript
import { Request, Response } from 'express';

@Get()
findAll(@Req() request: Request, @Res() response: Response) {
  return response.status(200).json({ message: 'Success' });
}
```

## 请求与响应

### 响应状态码

```typescript
@Post()
@HttpCode(201)  // 自定义状态码
create(@Body() createUserDto: CreateUserDto) {}

@Post()
@HttpCode(HttpStatus.CREATED)  // 使用枚举
create() {}
```

### 响应头

```typescript
@Post()
@Header('Cache-Control', 'none')
@Header('X-Custom-Header', 'value')
create() {}
}
```

### 重定向

```typescript
@Get()
@Redirect('https://nestjs.com', 301)
redirect() {}

@Get('docs')
@Redirect('https://docs.nestjs.com', 302)
getDocs(@Query('version') version) {
  if (version && version === '5') {
    return { url: 'https://docs.nestjs.com/v5/' };
  }
}
```

## 资源路由

CRUD 控制器示例：

```typescript
@Controller('users')
export class UsersController {
  constructor(private readonly usersService: UsersService) {}

  @Post()
  create(@Body() createUserDto: CreateUserDto) {
    return this.usersService.create(createUserDto);
  }

  @Get()
  findAll(@Query() paginationQuery: PaginationQueryDto) {
    return this.usersService.findAll(paginationQuery);
  }

  @Get(':id')
  findOne(@Param('id') id: string) {
    return this.usersService.findOne(id);
  }

  @Patch(':id')
  update(@Param('id') id: string, @Body() updateUserDto: UpdateUserDto) {
    return this.usersService.update(id, updateUserDto);
  }

  @Delete(':id')
  remove(@Param('id') id: string) {
    return this.usersService.remove(id);
  }
}
```
