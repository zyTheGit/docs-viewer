## 架构理念

### MVC 架构

NestJS 采用经典的 MVC（Model-View-Controller）架构：

```
┌─────────────────────────────────────────┐
│              Client Request              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│            Controller Layer              │
│  - 接收请求                              │
│  - 参数验证                              │
│  - 调用服务                              │
│  - 返回响应                              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│              Service Layer               │
│  - 业务逻辑                              │
│  - 数据处理                              │
│  - 事务管理                              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│          Repository/Model Layer          │
│  - 数据访问                              │
│  - ORM 映射                              │
│  - 数据库操作                            │
└─────────────────────────────────────────┘
```

### 分层架构

```typescript
// Controller - 控制器层
@Controller('users')
export class UsersController {
  constructor(private readonly usersService: UsersService) {}

  @Get()
  findAll() {
    return this.usersService.findAll();
  }
}

// Service - 服务层
@Injectable()
export class UsersService {
  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
  ) {}

  findAll(): Promise<User[]> {
    return this.userRepository.find();
  }
}

// Entity - 实体层
@Entity('users')
export class User {
  @PrimaryGeneratedColumn()
  id: number;

  @Column()
  name: string;
}
```

### 模块化设计

NestJS 通过模块组织应用：

```
AppModule (根模块)
├── AuthModule (认证模块)
│   ├── AuthController
│   ├── AuthService
│   ├── JwtStrategy
│   └── AuthGuard
├── UserModule (用户模块)
│   ├── UserController
│   ├── UserService
│   └── User
└── PostModule (文章模块)
    ├── PostController
    ├── PostService
    └── Post
```

## 依赖注入

### 什么是依赖注入？

依赖注入（Dependency Injection, DI）是一种设计模式，用于实现控制反转（IoC）。

**传统方式（不推荐）：**

```typescript
class UsersController {
  private usersService: UsersService;
  
  constructor() {
    this.usersService = new UsersService(); // 直接创建依赖
  }
}
```

**依赖注入方式（推荐）：**

```typescript
@Controller('users')
export class UsersController {
  constructor(private readonly usersService: UsersService) {}
  // Nest 自动注入 UsersService 实例
}
```

### 提供者（Providers）

#### 基本提供者

```typescript
// users.service.ts
import { Injectable } from '@nestjs/common';

@Injectable()  // 标记为可注入的提供者
export class UsersService {
  private users: User[] = [];

  create(user: User) {
    this.users.push(user);
  }

  findAll(): User[] {
    return this.users;
  }
}
```

```typescript
// users.module.ts
@Module({
  providers: [UsersService],  // 注册提供者
  controllers: [UsersController],
})
export class UsersModule {}
```

#### 类提供者

```typescript
@Module({
  providers: [UsersService],
  // 简写形式，等同于：
  providers: [
    {
      provide: UsersService,
      useClass: UsersService,
    },
  ],
})
export class UsersModule {}
```

#### 值提供者

```typescript
import { CACHE_MANAGER } from '@nestjs/cache-manager';

const cacheManager = {
  get: () => {},
  set: () => {},
};

@Module({
  providers: [
    {
      provide: CACHE_MANAGER,
      useValue: cacheManager,
    },
  ],
})
export class AppModule {}
```

#### 工厂提供者

```typescript
@Module({
  providers: [
    {
      provide: 'DATABASE_CONNECTION',
      useFactory: (configService: ConfigService) => {
        return createConnection({
          host: configService.get('DB_HOST'),
          port: configService.get('DB_PORT'),
        });
      },
      inject: [ConfigService],  // 注入依赖
    },
  ],
})
export class AppModule {}
```

#### 异步工厂提供者

```typescript
@Module({
  providers: [
    {
      provide: 'ASYNC_CONNECTION',
      useFactory: async (configService: ConfigService) => {
        const connection = await createConnection({
          host: configService.get('DB_HOST'),
        });
        return connection;
      },
      inject: [ConfigService],
    },
  ],
})
export class AppModule {}
```

#### 别名提供者

```typescript
abstract class Logger {
  abstract log(message: string): void;
}

@Injectable()
class ConsoleLogger extends Logger {
  log(message: string) {
    console.log(message);
  }
}

@Module({
  providers: [
    ConsoleLogger,
    {
      provide: Logger,
      useExisting: ConsoleLogger,  // 使用现有提供者
    },
  ],
})
export class AppModule {}
```

### 注入方式

#### 构造函数注入（推荐）

```typescript
@Controller('users')
export class UsersController {
  constructor(
    private readonly usersService: UsersService,
    private readonly authService: AuthService,
  ) {}
}
```

#### 属性注入

```typescript
@Controller('users')
export class UsersController {
  @Inject(UsersService)
  private readonly usersService: UsersService;
}
```

#### 可选注入

```typescript
import { Optional } from '@nestjs/common';

@Controller('users')
export class UsersController {
  constructor(
    @Optional() 
    private readonly logger?: Logger,
  ) {
    // logger 可能为 null
  }
}
```

#### 属性注入与标记

```typescript
export const DATABASE_CONNECTION = 'DATABASE_CONNECTION';

@Injectable()
export class UsersService {
  constructor(
    @Inject(DATABASE_CONNECTION)
    private connection: Connection,
  ) {}
}
```

### 循环依赖

#### 使用 forwardRef 解决循环依赖

```typescript
// users.service.ts
@Injectable()
export class UsersService {
  constructor(
    @Inject(forwardRef(() => AuthService))
    private authService: AuthService,
  ) {}
}

// auth.service.ts
@Injectable()
export class AuthService {
  constructor(
    @Inject(forwardRef(() => UsersService))
    private usersService: UsersService,
  ) {}
}
```

#### 模块级别的 forwardRef

```typescript
@Module({
  imports: [forwardRef(() => AuthModule)],
  providers: [UsersService],
  exports: [UsersService],
})
export class UsersModule {}
```

## 生命周期

NestJS 提供了完整的生命周期钩子：

```
┌─────────────────────────────────────────┐
│            Application Start             │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│         onModuleInit()                   │
│  模块初始化后调用                        │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│         onApplicationBootstrap()         │
│  应用启动完成后调用                      │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│          Application Running             │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│         beforeApplicationShutdown()      │
│  应用关闭前调用                          │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│         onModuleDestroy()                │
│  模块销毁后调用                          │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│         onApplicationShutdown()          │
│  应用完全关闭后调用                      │
└─────────────────────────────────────────┘
```

### 实现生命周期钩子

```typescript
import {
  Injectable,
  OnModuleInit,
  OnModuleDestroy,
  BeforeApplicationShutdown,
  OnApplicationBootstrap,
  OnApplicationShutdown,
} from '@nestjs/common';

@Injectable()
export class UsersService
  implements
    OnModuleInit,
    OnApplicationBootstrap,
    OnModuleDestroy,
    BeforeApplicationShutdown,
    OnApplicationShutdown
{
  onModuleInit() {
    console.log('UsersService: Module initialized');
  }

  onApplicationBootstrap() {
    console.log('UsersService: Application bootstrapped');
  }

  beforeApplicationShutdown(signal?: string) {
    console.log(`UsersService: Before shutdown (signal: ${signal})`);
  }

  onModuleDestroy() {
    console.log('UsersService: Module destroyed');
  }

  onApplicationShutdown(signal?: string) {
    console.log(`UsersService: Application shutdown (signal: ${signal})`);
  }
}
```

### 实际应用场景

```typescript
@Injectable()
export class DatabaseService implements OnModuleInit, OnModuleDestroy {
  private connection: Connection;

  async onModuleInit() {
    // 初始化数据库连接
    this.connection = await createConnection();
    console.log('Database connected');
  }

  async onModuleDestroy() {
    // 关闭数据库连接
    if (this.connection) {
      await this.connection.close();
      console.log('Database disconnected');
    }
  }
}
```

```typescript
@Injectable()
export class CacheService implements OnApplicationBootstrap, OnApplicationShutdown {
  private redisClient: RedisClient;

  async onApplicationBootstrap() {
    // 应用启动后初始化缓存
    this.redisClient = new RedisClient();
    await this.redisClient.connect();
  }

  async onApplicationShutdown() {
    // 应用关闭前清理缓存
    await this.redisClient.quit();
  }
}
```

## 模块系统概览

### 模块定义

```typescript
import { Module } from '@nestjs/common';
import { UsersController } from './users.controller';
import { UsersService } from './users.service';

@Module({
  imports: [],           // 导入其他模块
  controllers: [UsersController],  // 控制器
  providers: [UsersService],       // 提供者
  exports: [UsersService],         // 导出服务
```

### 模块属性

| 属性 | 说明 | 示例 |
|------|------|------|
| imports | 导入其他模块 | `[AuthModule, TypeOrmModule]` |

## 请求生命周期

### 完整请求流程

```
Incoming Request
       │
       ▼
┌─────────────────────────────────────────┐
│            Middleware                     │
│  - 日志记录                              │
│  - 请求预处理                            │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│              Guard                       │
│  - 认证检查                              │
│  - 权限验证                              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│        Interceptor (Before)              │
│  - 请求日志                              │
│  - 数据转换                              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│              Pipe                        │
│  - 参数验证                              │
│  - 数据转换                              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│          Controller Handler              │
│  - 业务逻辑处理                          │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│        Interceptor (After)               │
│  - 响应转换                              │
│  - 响应日志                              │
└────────────┬────────────────────────────┘
             │
             ▼
┌─────────────────────────────────────────┐
│        Exception Filter                  │
│  - 异常捕获（如果发生错误）              │
└────────────┬────────────────────────────┘
             │
             ▼
      Outgoing Response
```

### 代码示例

```typescript
// 1. Middleware
@Injectable()
export class LoggerMiddleware implements NestMiddleware {
  use(req: Request, res: Response, next: Function) {
    console.log('Request...', req.url);
    next();
  }
}

// 2. Guard
@Injectable()
export class AuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    return request.headers.authorization;
  }
}

// 3. Interceptor
@Injectable()
export class LoggingInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    console.log('Before...');
    const now = Date.now();
    return next
      .handle()
      .pipe(
        tap(() => console.log(`After... ${Date.now() - now}ms`)),
      );
  }
}

// 4. Pipe
@Injectable()
export class ValidationPipe implements PipeTransform {
  transform(value: any, metadata: ArgumentMetadata) {
    console.log('Validating...', value);
    return value;
  }
}

// 5. Controller
@Controller('users')
@UseGuards(AuthGuard)
@UseInterceptors(LoggingInterceptor)
export class UsersController {
  constructor(private readonly usersService: UsersService) {}

  @Get()
  @UsePipes(ValidationPipe)
  findAll() {
    return this.usersService.findAll();
  }
}

// 6. Exception Filter
@Catch(HttpException)
export class HttpExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const status = exception.getStatus();

    response.status(status).json({
      statusCode: status,
      timestamp: new Date().toISOString(),
    });
  }
}
```

## 设计模式

### 1. 单例模式

NestJS 默认使用单例模式管理提供者：

```typescript
@Injectable()
export class UsersService {
  private users: User[] = [];  // 所有组件共享同一个实例
  
  create(user: User) {
    this.users.push(user);
  }
}
```

### 2. 工厂模式

```typescript
interface LoggerService {
  log(message: string): void;
}

@Injectable()
class ConsoleLogger implements LoggerService {
  log(message: string) {
    console.log(message);
  }
}

@Injectable()
class FileLogger implements LoggerService {
  log(message: string) {
    // 写入文件
  }
}

@Module({
  providers: [
    {
      provide: 'LOGGER',
      useFactory: (configService: ConfigService) => {
        return configService.get('LOG_TYPE') === 'file' 
          ? new FileLogger() 
          : new ConsoleLogger();
      },
      inject: [ConfigService],
    },
  ],
})
export class AppModule {}
```

### 3. 策略模式

```typescript
// strategies/payment.strategy.ts
export interface PaymentStrategy {
  pay(amount: number): Promise<void>;
}

@Injectable()
export class StripePaymentStrategy implements PaymentStrategy {
  async pay(amount: number) {
    // Stripe 支付逻辑
  }
}

@Injectable()
export class PayPalPaymentStrategy implements PaymentStrategy {
  async pay(amount: number) {
    // PayPal 支付逻辑
  }
}

// services/payment.service.ts
@Injectable()
export class PaymentService {
  private strategies = new Map<string, PaymentStrategy>();

  registerStrategy(name: string, strategy: PaymentStrategy) {
    this.strategies.set(name, strategy);
  }

  async pay(strategyName: string, amount: number) {
    const strategy = this.strategies.get(strategyName);
    if (!strategy) {
      throw new Error(`Strategy ${strategyName} not found`);
    }
    return strategy.pay(amount);
  }
}
```

### 4. 装饰器模式

```typescript
// 自定义装饰器
export function Log() {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor,
  ) {
    const originalMethod = descriptor.value;

    descriptor.value = function (...args: any[]) {
      console.log(`Calling ${propertyKey} with args: ${JSON.stringify(args)}`);
      const result = originalMethod.apply(this, args);
      console.log(`Result: ${JSON.stringify(result)}`);
      return result;
    };

    return descriptor;
  };
}

// 使用装饰器
@Injectable()
export class UsersService {
  @Log()
  findAll(): User[] {
    return this.users;
  }
}
```

### 5. 观察者模式

```typescript
// events/user-created.event.ts
export class UserCreatedEvent {
  constructor(
    public readonly userId: number,
    public readonly email: string,
  ) {}
}

// events/user-created.listener.ts
@EventsHandler(UserCreatedEvent)
export class UserCreatedListener implements IEventHandler<UserCreatedEvent> {
  handle(event: UserCreatedEvent) {
    console.log(`User created: ${event.userId}`);
    // 发送欢迎邮件等
  }
}

// users.service.ts
@Injectable()
export class UsersService {
  constructor(private eventEmitter: EventEmitter2) {}

  async create(user: CreateUserDto) {
    const newUser = await this.userRepository.save(user);
    
    // 触发事件
    this.eventEmitter.emit(
      'user.created',
      new UserCreatedEvent(newUser.id, newUser.email),
    );
    
    return newUser;
  }
}
```

## 最佳实践

### 1. 服务职责单一

```typescript
// ❌ 不好：服务职责过多
@Injectable()
export class UsersService {
  createUser() {}
  sendEmail() {}
  validateUser() {}
  generateToken() {}
}

// ✅ 好：职责分离
@Injectable()
export class UsersService {
  constructor(
    private readonly emailService: EmailService,
    private readonly authService: AuthService,
  ) {}
  
  createUser() {}
}

@Injectable()
export class EmailService {
  sendEmail() {}
}

@Injectable()
export class AuthService {
  validateUser() {}
  generateToken() {}
}
```
