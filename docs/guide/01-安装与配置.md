## 环境要求

### 必需环境

| 工具 | 版本要求 | 说明 |
|------|----------|------|
| Node.js | >= 18.x | 推荐 LTS 版本 |
| npm | >= 9.x | 或 yarn >= 1.22.x / pnpm >= 8.x |
| TypeScript | >= 5.x | Nest 自动安装 |

### 推荐工具

- **VS Code** - 代码编辑器
- **Postman** - API 测试工具
- **Database Tool** - 数据库管理工具（如 DBeaver、TablePlus）

### 检查环境

```bash
# 检查 Node.js 版本
node -v
# 输出: v18.17.0 或更高

# 检查 npm 版本
npm -v
# 输出: 9.6.7 或更高

# 检查 TypeScript 版本
tsc -v
# 输出: Version 5.0.0 或更高
```

### 使用 nvm 管理 Node.js（推荐）

```bash
# 安装 nvm（macOS/Linux）
curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.0/install.sh | bash

# 安装最新 LTS 版本
nvm install --lts

# 切换 Node.js 版本
nvm use 18

# 设置默认版本
nvm alias default 18
```

## 安装 Nest CLI

### 全局安装

```bash
# 使用 npm
npm install -g @nestjs/cli

# 使用 yarn
yarn global add @nestjs/cli

# 使用 pnpm
pnpm add -g @nestjs/cli

# 验证安装
nest -v
# 输出: 10.x.x
```

### CLI 常用命令

```bash
# 查看帮助
nest --help

# 查看命令详细帮助
nest new --help

# 查看版本
nest -v
nest --version
```

## 创建项目

### 方式一：使用 CLI（推荐）

```bash
# 创建新项目
nest new project-name

# 选择包管理器
? Which package manager would you ❤️ to use? 
  npm
  yarn
  pnpm
```

### 方式二：使用 Git 克隆

```bash
# 克隆 TypeScript 版本
git clone https://github.com/nestjs/typescript-starter.git project-name
cd project-name
npm install

# 克隆 JavaScript 版本
git clone https://github.com/nestjs/javascript-starter.git project-name
cd project-name
npm install
```

### 方式三：手动创建

```bash
# 创建目录
mkdir project-name && cd project-name

# 初始化 package.json
npm init -y

# 安装依赖
npm install @nestjs/common @nestjs/core @nestjs/platform-express rxjs reflect-metadata

# 安装开发依赖
npm install -D @nestjs/cli @types/node typescript

# 创建必要文件
mkdir src
touch src/main.ts src/app.module.ts src/app.controller.ts src/app.service.ts
touch tsconfig.json nest-cli.json
```

### 项目启动

```bash
# 开发模式（支持热重载）
npm run start:dev

# 生产模式
npm run start:prod

# 调试模式
npm run start:debug

# 普通启动
npm run start
```

访问 http://localhost:3000 查看应用。

### 核心文件说明

#### main.ts - 应用入口

```typescript
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  await app.listen(3000);
}
bootstrap();
```

**配置选项：**

```typescript
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ValidationPipe } from '@nestjs/common';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, {
    logger: ['error', 'warn', 'log'],  // 日志级别
    cors: true,                         // 启用 CORS
    bodyParser: true,                   // 启用 body 解析
  });

  // 全局验证管道
  app.useGlobalPipes(new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
  }));

  // 设置全局前缀
  app.setGlobalPrefix('api');

  // 启用 CORS
  app.enableCors({
    origin: ['http://localhost:3000'],
    credentials: true,
  });

  // 使用 Fastify（可选）
  // const app = await NestFactory.create<NestFastifyApplication>(
  //   AppModule,
  //   new FastifyAdapter()
  // );

  await app.listen(3000);
}
bootstrap();
```

#### app.module.ts - 根模块

```typescript
import { Module } from '@nestjs/common';
import { AppController } from './app.controller';
import { AppService } from './app.service';

@Module({
  imports: [],          // 导入其他模块
  controllers: [AppController],  // 控制器
  providers: [AppService],       // 提供者
  exports: [],          // 导出服务
})
export class AppModule {}
```

#### app.controller.ts - 根控制器

```typescript
import { Controller, Get } from '@nestjs/common';
import { AppService } from './app.service';

@Controller()
export class AppController {
  constructor(private readonly appService: AppService) {}

  @Get()
  getHello(): string {
    return this.appService.getHello();
  }
}
```

#### app.service.ts - 根服务

```typescript
import { Injectable } from '@nestjs/common';

@Injectable()
export class AppService {
  getHello(): string {
    return 'Hello World!';
  }
}
```

### 使用配置

```typescript
// src/app.module.ts
import { ConfigModule } from '@nestjs/config';
import configuration from './config';

@Module({
  imports: [
    ConfigModule.forRoot({
      isGlobal: true,        // 全局模块
      load: [configuration], // 加载配置
      envFilePath: [`.env.${process.env.NODE_ENV || 'development'}`, '.env'],
      cache: true,           // 缓存配置
    }),
  ],
})
export class AppModule {}
```

```typescript
// 在服务中使用
import { ConfigService } from '@nestjs/config';
import { Injectable } from '@nestjs/common';

@Injectable()
export class AppService {
  constructor(private configService: ConfigService) {}

  getDatabaseConfig() {
    return {
      host: this.configService.get<string>('database.host'),
      port: this.configService.get<number>('database.port'),
    };
  }

  getAppPort(): number {
    return this.configService.get<number>('app.port');
  }
}
```

## 常见问题

### 1. 端口被占用

**问题：** `Error: listen EADDRINUSE: address already in use :::3000`

**解决方案：**

```bash
# 查找占用端口的进程
lsof -i :3000

# 杀死进程
kill -9 <PID>

# 或修改端口
# src/main.ts
await app.listen(3001);
```

### 2. 依赖安装失败

**问题：** `npm install` 失败

**解决方案：**

```bash
# 清除缓存
npm cache clean --force

# 删除 node_modules
rm -rf node_modules package-lock.json

# 重新安装
npm install

# 或使用 yarn
yarn install
```

### 3. TypeScript 编译错误

**问题：** 类型错误

**解决方案：**

```bash
# 检查 TypeScript 版本
tsc -v

# 更新依赖
npm update

# 重新构建
npm run build
```

### 4. 热重载不工作

**问题：** 修改代码后不自动重载

**解决方案：**

```bash
# 确保使用正确的启动命令
npm run start:dev

# 检查 nest-cli.json 配置
{
  "compilerOptions": {
    "watchAssets": true
  }
}
```

### 5. 环境变量不生效

**问题：** `process.env.XXX` 为 undefined

**解决方案：**

```typescript
// 确保 ConfigModule.forRoot() 的 isGlobal: true
ConfigModule.forRoot({
  isGlobal: true,
})

// 检查 .env 文件是否在项目根目录
// 确保环境变量名称正确
console.log(process.env); // 调试所有环境变量
```
